def inquire_unauthorized_loan(loan_date=None, loan_amount=None):
    """用户质疑存在未经授权的借款记录"""
    # 使用标准库函数替代未定义的辅助函数
    loan_date_parsed = None
    if loan_date:
        if isinstance(loan_date, str):
            # 尝试解析日期字符串（仅支持部分格式）
            for fmt in ["%Y年%m月%d日", "%Y年%m月"]:
                try:
                    from datetime import datetime
                    loan_date_parsed = datetime.strptime(loan_date, fmt).date()
                    break
                except ValueError:
                    continue

    loan_amount_parsed = None
    if loan_amount:
        if isinstance(loan_amount, str):
            # 提取金额数字部分
            import re
            amount_match = re.search(r"[\d\.]+", loan_amount)
            if amount_match:
                try:
                    loan_amount_parsed = float(amount_match.group())
                except ValueError:
                    pass

    unauthorized = False
    severity = "低"

    if loan_date_parsed and loan_amount_parsed:
        # 模拟调用系统接口验证是否为用户授权
        # 这里简化为根据金额判断是否为高风险未经授权贷款
        if loan_amount_parsed > 10000:
            unauthorized = True
            severity = "高"
        elif loan_amount_parsed > 500:
            unauthorized = True
            severity = "中"
        else:
            unauthorized = False

    return {
        "status": "success",
        "unauthorized": unauthorized,
        "severity": severity if unauthorized else "无",
    }

def refuse_to_use_service(credit_limit_status=None):
    """用户因系统显示问题不敢使用服务"""
    status = "success"
    handling_priority = "低"
    if credit_limit_status and "临时提高" in str(credit_limit_status):
        handling_priority = "中"
        if "高价值" in str(credit_limit_status) or "异常" in str(credit_limit_status):
            handling_priority = "高"
    return {
        "status": status,
        "handling_priority": handling_priority
    }
